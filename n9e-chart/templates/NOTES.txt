  查看 Gateway 状态: kubectl get gateway --namespace {{ $namespace }}
  查看 HTTPRoute 状态: kubectl get httproute --namespace {{ $namespace }}
{{- else if eq .Values.service.type "NodePort" }}
获取 NodePort 访问地址：
  export NODE_IP=$(kubectl get nodes --namespace {{ $namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  export NODE_PORT=$(kubectl get svc --namespace {{ $namespace }} {{ $fullname }} -o jsonpath="{.spec.ports[0].nodePort}")
  echo "访问地址: http://$NODE_IP:$NODE_PORT"
{{- else if eq .Values.service.type "LoadBalancer" }}
获取 LoadBalancer 外部 IP：
  kubectl get svc --namespace {{ $namespace }} {{ $fullname }} -w
{{- else }}
使用端口转发访问：
  kubectl port-forward --namespace {{ $namespace }} svc/{{ $fullname }} {{ .Values.service.port }}:{{ .Values.service.port }}
  然后访问: http://127.0.0.1:{{ .Values.service.port }}
{{- end }}

2. 常用运维命令
------------------------------------------------------------------------
查看 Pod 状态：
  kubectl get pods --namespace {{ $namespace }} -l app.kubernetes.io/instance={{ .Release.Name }}

查看日志：
  kubectl logs --namespace {{ $namespace }} -l app.kubernetes.io/instance={{ .Release.Name }} -f

手动水平扩展副本数：
  kubectl scale deployment --namespace {{ $namespace }} {{ $fullname }} --replicas=<N>

升级：
  helm upgrade {{ .Release.Name }} <chart> --namespace {{ $namespace }} -f values.yaml

卸载：
  helm uninstall {{ .Release.Name }} --namespace {{ $namespace }}
